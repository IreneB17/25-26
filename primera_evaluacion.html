int tamOso = 50;


int escala(int valor, int size) {
  return valor * size / 50;
}

// función oso
boolean oso(int x, int y, int size) {
  // OREJAS
  ellipse(x-escala(60, size), y-escala(60, size), escala(70, size), escala(70, size));
  ellipse(x+escala(60, size), y-escala(60, size), escala(70, size), escala(70, size));

  // CABEZA
  ellipse(x, y, escala(150, size), escala(150, size));

  // OJOS
  ellipse(x-escala(20, size), y-escala(20, size), escala(30, size), escala(50, size));
  ellipse(x-escala(10, size), y-escala(20, size), escala(5, size), escala(5, size));
  ellipse(x+escala(20, size), y-escala(20, size), escala(30, size), escala(50, size));
  ellipse(x+escala(10, size), y-escala(20, size), escala(5, size), escala(5, size));

  // NARIZ
  ellipse(x, y+escala(10, size), escala(30, size), escala(10, size));

  // BOCA
  ellipse(x, y+escala(40, size), escala(15, size), escala(5, size));

  return true;
}


void dibuja() {
  
  background(0,0,0);

  
  if (keyPressed) {
    if (key == "g") {     // tecla g  más grande
      tamOso = tamOso + 5;
      if (tamOso > 200) { // límite máximo permitido
        tamOso = 200;
      }
    }
    if (key == "p") {     // tecla p  más pequeño
      tamOso = tamOso - 5;
      if (tamOso < 10) {  // límite mínimo permitido
        tamOso = 10;
      }
    }
  }

  // click derecho  dibujar oso en el lugar del mouse
  if (mousePressed && mouseButton == RIGHTBUTTON) {
    oso(mouseX, mouseY, tamOso);
  }
}

void main() {
  animate(dibuja);
}

Trabajo 5 - Semáforo

Semáforo temporizado 


int estado = 0;
int tiempoinicio;
void semaforo(){
  // SEMAFORO
  fill(40,140,10);
  rect(120,70,80,160);
  //LUCES ENCENDIDAS
  fill(230,20,20);
  fill(255,255,0);
  fill(60,255,0);
  //LUCES APAGADAS
  fill(128, 15, 15);
  fill(146, 146, 5);
  fill(39, 140, 8);
  //POSICIONES
  int x = 160;
  int r = 100;
  int a = 150;
  int v = 200;
  int d = 40;
  
  if(estado == 0){
    //ROJO
    fill(230,20,20);
    ellipse(x, r, d, d);
    fill(146, 146, 5);
    ellipse(x, a, d, d);
    fill(39, 140, 8);
    ellipse(x, v, d, d);
  } else if(estado == 1){
    //VERDE
    fill(128,15,15);
    ellipse(x,r,d,d);
    fill(146,146,5);
    ellipse(x,a,d,d);
    fill(60,255,0);
    ellipse(x,v,d,d);
  } else if(estado == 2){
    //AMARILLO
    fill(128,15,15);
    ellipse(x,r,d,d);
    fill(255,255,0);
    ellipse(x,a,d,d);
    fill(39,140,8);
    ellipse(x,v,d,d);
  }
  
  //CAMBIO DE COLOR
  int tiempoactual = millis();
  int duracion;
  
  if(estado == 0) duracion = 5000;
  else if(estado == 1) duracion = 3000;
  else duracion = 1500;
  
  if(tiempoactual - tiempoinicio > duracion){
    estado = (estado + 1)%3;
    tiempoinicio = tiempoactual;
  }
}

void main(){
  tiempoinicio = millis();
  animate(semaforo);
}
